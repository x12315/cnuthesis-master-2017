\chapter{GNSS/INS误差模型与观测算法}

本章围绕 GNSS/INS 组合导航滤波器的建模过程展开，目标是给出后续算法实现所需的统一状态空间表达。首先定义松耦合框架下的误差状态向量，并给出 INS 误差动力学方程；随后将连续时间模型整理为矩阵形式，明确系统矩阵与噪声驱动项；在此基础上，构建包含杆臂改正的松耦合观测模型\cite{yan2019strapdown,9106151}。上述建模结果将作为后续鲁棒滤波与参数优化方法设计的基础。

\section{误差模型的系统动态模型}

本节首先建立滤波器误差状态，再据此推导系统动力学方程和矩阵化表达，为后续量测更新与离散实现提供基础。

针对 GNSS/INS 松耦合（LC）架构，采用如下 15 维误差状态向量\cite{yan2019strapdown}：
\begin{equation}
    \delta \bm{x}_\text{LC} = [\delta \bm{r}^n, \delta \bm{v}^n, \bm{\phi}, \bm{\varepsilon}^b, \nabla^b]^T
    \label{eq:error_state}
\end{equation}
式中：上标 $n$ 表示导航坐标系，$b$ 表示载体坐标系。本文默认导航坐标系采用 ENU（东-北-天）约定；若个别公式沿用参考文献记号，仅视为符号习惯，不改变其物理含义。
$\delta \bm{r}^n = [\delta r_E, \delta r_N, \delta r_U]^T$ 为东向、北向、天向位置误差向量；
$\delta \bm{v}^n = [\delta v_E, \delta v_N, \delta v_U]^T$ 为东向、北向、天向速度误差向量；
$\bm{\phi} = [\phi_E, \phi_N, \phi_U]^T$ 为东向、北向、天向失准角向量；
$\bm{\varepsilon}^b = [\varepsilon_X, \varepsilon_Y, \varepsilon_Z]^T$ 为陀螺零偏；
$\nabla^b = [\nabla_X, \nabla_Y, \nabla_Z]^T$ 为加速度计零偏。

\section{松耦合误差动力学模型与矩阵形式}
基于捷联惯导误差微扰理论，并结合 IMU 随机误差的一阶马尔可夫建模，可得到惯性导航系统（INS）误差状态动力学模型\cite{yan2019strapdown,9106151}：
\begin{equation}
    \begin{cases}
        \delta \dot{\bm{r}}^n = -\bm{\omega}_{en}^n \times \delta \bm{r}^n + \delta \bm{v}^n                                                           \\[6pt]
        \delta \dot{\bm{v}}^n = \bm{C}_b^n \bm{f}^b \times \bm{\phi} - \left(2\bm{\omega}_{ie}^n + \bm{\omega}_{en}^n\right) \times \delta \bm{v}^n    \\[6pt]
        \quad\quad\quad\quad -\left(2\delta\bm{\omega}_{ie}^n + \delta\bm{\omega}_{en}^n\right) \times \bm{v}^n +\delta \bm{g}^n + \bm{C}_b^n \nabla^b \\[6pt]
        \dot{\bm{\phi}} = -\bm{\omega}_{in}^n \times \bm{\phi} + \delta\bm{\omega}_{in}^n - \bm{C}_b^n \bm{\varepsilon}^b                              \\[6pt]
        \dot{\bm{\varepsilon}}^b = -\dfrac{1}{T_\varepsilon} \bm{\varepsilon}^b + \bm{w}_\varepsilon                                                   \\[6pt]
        \dot{\nabla}^b = -\dfrac{1}{T_\nabla} \nabla^b + \bm{w}_\nabla
    \end{cases}
    \label{eq:ins_error_model}
\end{equation}
式中：$\bm{\omega}_{ie}^n$ 为地球自转角速度在导航系下的投影；$\bm{\omega}_{en}^n$ 为导航系相对地球坐标系的转动角速度；$\bm{\omega}_{in}^n = \bm{\omega}_{ie}^n + \bm{\omega}_{en}^n$；
$\bm{C}_b^n$ 为载体坐标系至导航坐标系的转换矩阵；
$T_\varepsilon$、$T_\nabla$ 分别为陀螺和加速度计相关时间；
$\bm{w}_\varepsilon$、$\bm{w}_\nabla$ 为驱动噪声，其统计特性可通过 Allan 方差分析或 IMU 产品指标估计\cite{wang2018using}；
误差向量 $\delta \bm{\omega}_{ie}^n$、$\delta \bm{\omega}_{en}^n$ 以及重力误差 $\delta \bm{g}^n$，
可表示为状态向量中位置误差 $\delta \bm{r}^n$、速度误差 $\delta \bm{v}^n$ 和失准角 $\bm{\phi}$ 的函数；
$\bm{\omega}_{ib}^b$ 和 $\bm{f}^b$ 分别表示陀螺仪与加速度计测量的角速度和比力。

为便于滤波器离散实现，式(\ref{eq:ins_error_model})可改写为标准状态空间形式：
\begin{equation}
    \delta \dot{\bm{x}}_k = \bm{F}_\text{SINS} \cdot \delta \bm{x}_k + \bm{G}_\text{SINS} \cdot \bm{w}_k
    \label{eq:matrix_dynamic}
\end{equation}


其中，$\delta \bm{x}_k$ 为第 $k$ 个离散时刻的 INS 误差状态向量；
$\bm{w}_k = [\bm{w}_\varepsilon^T, \bm{w}_\nabla^T]^T$ 为过程噪声，通常假设
$\bm{w}_k \sim N(0, \bm{Q}_k)$；$\bm{F}_{\text{SINS}}$ 为系统动力学矩阵。
采用指数离散化时，小时间间隔 $[t_{k-1}, t_k]$ 上的状态转移矩阵可写为
\[
    \Phi_{k,k-1} = e^{F \cdot (t_k - t_{k-1})};
\]
$\bm{G}_{\text{SINS}}$ 为噪声驱动矩阵，用于建立误差状态与噪声源之间的映射关系。

$\bm{F}_\text{SINS}$ 可进一步写作多个子矩阵的组合形式：
% 系统状态转移矩阵 F_SINS
\begin{equation}
    \bm{F}_\text{SINS} =
    \begin{bmatrix}
        \bm{M}_{rr}       & \bm{I}_{3\times3} & \bm{0}_{3\times3} & \bm{0}_{3\times3}                          & \bm{0}_{3\times3}                     \\[6pt]
        \bm{M}_{vr}       & \bm{M}_{vv}       & \bm{M}_{v\phi}    & \bm{0}_{3\times3}                          & \bm{C}_b^n                            \\[6pt]
        \bm{M}_{\phi r}   & \bm{M}_{\phi v}   & \bm{M}_{\phi\phi} & -\bm{C}_b^n                                & \bm{0}_{3\times3}                     \\[6pt]
        \bm{0}_{3\times3} & \bm{0}_{3\times3} & \bm{0}_{3\times3} & -\dfrac{1}{T_\varepsilon}\bm{I}_{3\times3} & \bm{0}_{3\times3}                     \\[6pt]
        \bm{0}_{3\times3} & \bm{0}_{3\times3} & \bm{0}_{3\times3} & \bm{0}_{3\times3}                          & -\dfrac{1}{T_\nabla}\bm{I}_{3\times3}
    \end{bmatrix}_{15\times15}
\end{equation}


\begin{equation}
    \begin{aligned}
        \bm{f}^n          & = \bm{C}_{b}^n\bm{f}_{b},                                                                                                               \\
        \bm{M}_{v\phi}    & = \bigl\lfloor \bm{f}^n \times \bigr\rfloor,                                                                                                    \\
        \bm{M}_{vv}       & = \bigl( \bm{v}^n \times \bigr)\bm{M}_{av} - \bigl\lfloor (2\bm{\omega}_{ie}^n+\bm{\omega}_{en}^n) \times \bigr\rfloor, \\
        \bm{M}_{vr}       & = \bigl( \bm{v}^n \times \bigr)(2\bm{M}_1+\bm{M}_2) + \bm{M}_3,                                                         \\[4pt]
        \bm{M}_{\phi\phi} & = -\bigl\lfloor \bm{\omega}_{in}^n \times \bigr\rfloor,                                                                                         \\
        \bm{M}_{\phi v}   & = \bm{M}_{av},                                                                                                                                  \\
        \bm{M}_{\phi r}   & = \bm{M}_1 + \bm{M}_2.
    \end{aligned}
\end{equation}


% M1
\begin{equation}
    \bm{M}_1 =
    \begin{bmatrix}
        0                  & 0 & 0 \\
        -\omega_{ie}\sin L & 0 & 0 \\
        \omega_{ie}\cos L  & 0 & 0
    \end{bmatrix}
\end{equation}

% M2
\begin{equation}
    \bm{M}_2 =
    \begin{bmatrix}
        0                           & 0 & \dfrac{v_N}{R_{Mh}^2}        \\[4pt]
        0                           & 0 & -\dfrac{v_E}{R_{Nh}^2}       \\[4pt]
        \dfrac{v_E\sec^2 L}{R_{Nh}} & 0 & -\dfrac{v_E\tan L}{R_{Nh}^2}
    \end{bmatrix}
\end{equation}

% Mav
\begin{equation}
    \bm{M}_{av} =
    \begin{bmatrix}
        0                      & -\dfrac{1}{R_{Mh}} & 0 \\[4pt]
        \dfrac{1}{R_{Nh}}      & 0                  & 0 \\[4pt]
        \dfrac{\tan L}{R_{Nh}} & 0                  & 0
    \end{bmatrix}
\end{equation}

% Mpv
\begin{equation}
    \bm{M}_{rv} =
    \begin{bmatrix}
        0                      & \dfrac{1}{R_{Mh}} & 0 \\[4pt]
        \dfrac{\sec L}{R_{Nh}} & 0                 & 0 \\[4pt]
        0                      & 0                 & 1
    \end{bmatrix}
\end{equation}

% Mpp
\begin{equation}
    \bm{M}_{rr} =
    \begin{bmatrix}
        0                               & 0 & -\dfrac{v_N}{R_{Mh}^2}       \\[4pt]
        \dfrac{v_E\sec L\tan L}{R_{Nh}} & 0 & -\dfrac{v_E\sec L}{R_{Nh}^2} \\[4pt]
        0                               & 0 & 0
    \end{bmatrix}
\end{equation}

其中 $L$ 表示纬度；$\bm{v}$ 与 $\bm{C}_{b}^n$ 可由 INS 机械编排结果获得。


\section{松耦合观测模型}

在 GNSS/INS 松耦合集成中，滤波器以 GNSS 输出的位置、速度作为外部参考，通过构造 INS 与 GNSS 解之间的差值观测，估计导航误差与惯性器件零偏。在时刻 $k$，量测向量 $\delta \bm{z}_k^{\text{LC}}$ 可写为

\begin{equation}
    \delta \bm{z}_k^{\text{LC}} =
    \begin{pmatrix}
        \delta \bm{z}_r \\[2pt]
        \delta \bm{z}_v
    \end{pmatrix}
    =
    \begin{pmatrix}
        \bm{r}_k^{\text{IMU}} - \bm{r}_k^{\text{GNSS}} \\[2pt]
        \bm{v}_k^{\text{IMU}} - \bm{v}_k^{\text{GNSS}}
    \end{pmatrix}.
    \label{eq:lc_meas_vector}
\end{equation}

由于 GNSS 天线与 IMU 通常不共点，需要考虑从 IMU 中心指向 GNSS 相位中心的杆臂向量 $\bm{e}^b$。经杆臂效应改正后，量测模型可表示为。

\begin{equation}
    \delta \bm{z}_k^{\text{LC}} = \bm{H}_k^{\text{LC}} \cdot \delta \bm{x}_k + \bm{e}_{\text{LC}},
\end{equation}

其中 $\bm{e}_{\text{LC}} = (\bm{e}_r, \bm{e}_v)^T$ 为 GNSS 位置和速度观测噪声，其协方差满足
$E[(\bm{e}_r^T, \bm{e}_{vi}^T)^T(\bm{e}_r^T, \bm{e}_{vk}^T)] = \bm{R}_k\delta_{ik}$，$\delta_{ik}$ 为克罗内克函数。对应设计矩阵 $\bm{H}_k^{\text{LC}}$ 为

\begin{equation}
    \delta \bm{z}_k^{\text{LC}} =
    \begin{pmatrix}
        \delta z_r \\[2pt]
        \delta z_v
    \end{pmatrix}
    =
    \begin{pmatrix}
        \bm{r}_1^{\text{IMU}} - \bm{r}_1^{\text{GNSS}} + \bm{C}_n^b \bm{e}^b \\[4pt]
        \bm{v}_1^{\text{IMU}} - \bm{v}_1^{\text{GNSS}} - (\bm{\omega}_n^i \times) \bm{C}_n^b \mathbf{e}^b - \bm{C}_n^b (\mathbf{e}^b \times) \bm{\omega}_{ib}^b
    \end{pmatrix},
\end{equation}

\begin{equation}
    \bm{H}_k^{\text{LC}} =
    \begin{pmatrix}
        \bm{I}_{3 \times 3} & \bm{0}_{3 \times 3} & \bigl(\bm{C}_{b}^n \bm{\ell}^b\bigr)\times & \bm{0}_{3 \times 3} & \bm{0}_{3 \times 3} \\
        \bm{0}_{3 \times 3} & \bm{I}_{3 \times 3} & - \left[\left(\bm{\omega}_{in}^n\times\right)\left(\bm{C}_b^n\ell^b\right)+\bm{C}_b^n\left(\ell^b\times\bm{\omega}_{ib}^b\right)\times \right] & -\bm{C}_b^n\left(\ell^b\times\right) & \bm{0}_{3 \times 3}
    \end{pmatrix}.
\end{equation}

基于上述系统模型与观测模型，后续章节将围绕复杂城市环境下的非高斯噪声与异常观测问题，进一步引入鲁棒滤波与自适应参数策略，以提升 MEMS-GNSS/INS 组合导航的精度与稳定性。

\section{本章小结}

本章围绕 GNSS/INS 组合导航滤波建模给出了完整的理论链路：首先定义了松耦合误差状态向量，随后推导了 INS 误差动力学方程，并将其整理为便于离散实现的系统矩阵形式；在此基础上，构建了考虑杆臂效应的松耦合观测模型。上述内容为下一章鲁棒滤波算法设计与实验验证提供了统一的建模基础。
